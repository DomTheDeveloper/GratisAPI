Quick start

It's just HTTP GET on static files. Use whatever you already have.

# curl — list every API
curl https://domthedeveloper.github.io/GratisAPI/api/index.json

# get one animal's full taxonomy
curl .../api/animals/lion.json

# all 118 chemical elements
curl .../api/elements/index.json
// fetch — a random quote in the browser
const res = await fetch(
  ".../api/quotes/index.json");
const { results } = await res.json();
const q = results[
  Math.floor(Math.random() * results.length)];
console.log(`"${q.quote}" — ${q.author}`);
